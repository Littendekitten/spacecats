🔥 Firebase Security Rules

Copy and paste these rules into your Firebase Console so your Realtime Database and Firestore function properly without permission errors.

Location in Console: Realtime Database → Rules tab

{
  "rules": {
    "players": {
      ".read": true,
      ".write": true
    },
    "projectiles": {
      ".read": true,
      ".write": true
    },
    "coins": {
      ".read": true,
      ".write": true
    }
  }
}

Location in Console: Firestore Database → Rules tab

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read, write: if true;
    }
  }
}
